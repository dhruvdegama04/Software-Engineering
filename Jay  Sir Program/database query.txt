8.forein key
CREATE TABLE product(
pid int(11) PRIMARY KEY AUTO_INCREMENT,
pname varchar(255) not null,
pprice int(11) not null,
pcategory varchar(255),
pdescription varchar(255),
sellerId int(11),
FOREIGN KEY (sellerId) REFERENCES seller(seller_id) on update cascade on delete cascade
);

CREATE OR REPLACE TRIGGER trg_update_balance
AFTER INSERT ON transactions
FOR EACH ROW
BEGIN

    IF :NEW.transaction_type = 'EXPENSE' THEN

        UPDATE users
        SET balance = balance - :NEW.amount
        WHERE id = :NEW.user_id;

    END IF;

END;

DECLARE

    CURSOR c_transactions IS
        SELECT user_id, amount
        FROM transactions;

BEGIN

    FOR transaction_record IN c_transactions LOOP

        DBMS_OUTPUT.PUT_LINE(
            'User: ' || transaction_record.user_id ||
            ' Amount: ' || transaction_record.amount

    END LOOP;

END;

Store Procedure
CREATE OR REPLACE PROCEDURE calculate_total_spend (
    p_user_id IN NUMBER
)
IS
    v_total_spend NUMBER;
BEGIN

    SELECT NVL(SUM(amount), 0)
    INTO v_total_spend
    FROM transactions
    WHERE user_id = p_user_id;

    DBMS_OUTPUT.PUT_LINE(
        'Total Spend = ' || v_total_spend
    );

EXCEPTION
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE(
            'Error: ' || SQLERRM
        );

END;
/
